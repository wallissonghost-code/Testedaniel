// ==UserScript==
// @name         Monitor v505 - Base corrigida
// @namespace    http://tampermonkey.net/
// @version      505.2
// @description  Monitora somente Aguardando entrega, pausa em Todos/Entregues e registra apenas em saídas reais
// @author       Daniel Alexandre
// @match        https://app.econdos.com.br/*
// @run-at       document-start
// @grant        none
// ==/UserScript==

(function () {
    "use strict";

    const STORAGE_HISTORICO = "monitor503_historico";
    const INTERVALO = 300;

    let unidadeAtual = "";
    let quantidadeAtual = 0;
    let sessaoAtiva = false;
    let filtroValido = false;
    let ultimaURL = location.href;
    let saidaProcessada = false;
    let logsSistema = [];
    let intervaloMatrix = null;

    function normalizarTexto(valor) {
        return String(valor || "")
            .trim()
            .toLowerCase()
            .normalize("NFD")
            .replace(/[\u0300-\u036f]/g, "");
    }

    function escaparHTML(valor) {
        return String(valor ?? "")
            .replaceAll("&", "&amp;")
            .replaceAll("<", "&lt;")
            .replaceAll(">", "&gt;")
            .replaceAll('"', "&quot;")
            .replaceAll("'", "&#039;");
    }

    function registrarEvento(tipo, mensagem, destaque = "") {
        logsSistema.unshift({
            tipo,
            mensagem,
            destaque,
            hora: new Date().toLocaleTimeString("pt-BR")
        });
        logsSistema = logsSistema.slice(0, 100);
        atualizarPainel();
    }

    function obterHistorico() {
        try {
            const lista = JSON.parse(
                localStorage.getItem(STORAGE_HISTORICO) || "[]"
            );
            return Array.isArray(lista) ? lista : [];
        } catch {
            return [];
        }
    }

    function gravarHistorico(lista) {
        localStorage.setItem(
            STORAGE_HISTORICO,
            JSON.stringify(lista.slice(0, 1000))
        );
    }

    function modoFiltro() {
        const botao = Array.from(
            document.querySelectorAll(
                'button[data-testid="delivery-status-filter-button"]'
            )
        ).find(el => el.offsetParent !== null);

        if (!botao) return "outro";

        const texto = normalizarTexto(
            botao.innerText ||
            botao.textContent ||
            botao.getAttribute("aria-label") ||
            botao.getAttribute("title") ||
            ""
        );

        if (texto.includes("aguardando") && texto.includes("entrega")) {
            return "aguardando";
        }
        if (texto.includes("todos") || texto === "todas") return "todos";
        if (texto.includes("entregue")) return "entregues";
        return "outro";
    }

    function lerUnidadeSelecionada() {
        const inputsVisiveis = Array.from(document.querySelectorAll("input"))
            .filter(input => input.offsetParent !== null && !input.disabled);

        // Prioriza o campo específico quando ele existir.
        const campoPreferido = inputsVisiveis.find(input =>
            input.getAttribute("data-testid") ===
            "residence-autocomplete-input-search"
        );

        if (campoPreferido) {
            const valor = String(campoPreferido.value || "").trim();
            if (/^\d+\/\d+$/.test(valor)) return valor;
        }

        // Fallback da versão Premium funcional: procura qualquer input
        // visível que contenha exatamente uma unidade no formato 2/202.
        for (const input of inputsVisiveis) {
            const valor = String(input.value || "").trim();
            if (/^\d+\/\d+$/.test(valor)) return valor;
        }

        return "";
    }

    function lerX() {
        const tabela = Array.from(
            document.querySelectorAll('[data-testid="delivery-table"]')
        ).find(el => el.offsetParent !== null);

        if (!tabela) return null;

        for (const elemento of tabela.querySelectorAll("footer small")) {
            if (elemento.offsetParent === null) continue;

            const match = normalizarTexto(elemento.textContent).match(
                /(?:^|\s)(\d+)\s+encomenda(?:\(s\)|s)?(?:\s|$)/
            );

            if (match) return Number(match[1]);
        }

        return null;
    }

    function salvarPendencia(motivo) {
        if (saidaProcessada) return false;
        if (!sessaoAtiva) return false;
        if (!filtroValido) return false;
        if (!unidadeAtual) return false;
        if (quantidadeAtual <= 0) return false;

        saidaProcessada = true;

        const texto = quantidadeAtual === 1
            ? "1 encomenda sem baixa"
            : `${quantidadeAtual} encomendas sem baixa`;

        const historico = obterHistorico().filter(item =>
            (item.codigo || item.apartamento) !== unidadeAtual
        );

        historico.unshift({
            codigo: unidadeAtual,
            apartamento: unidadeAtual,
            quantidade: quantidadeAtual,
            motivo: `${texto} · ${motivo}`,
            dataCompleta: new Date().toLocaleString("pt-BR"),
            timestamp: Date.now()
        });

        gravarHistorico(historico);
        registrarEvento("HISTÓRICO", `${texto} · ${motivo}`, unidadeAtual);
        return true;
    }

    function removerDoHistorico(unidade) {
        if (!unidade) return false;

        const historico = obterHistorico();
        const novo = historico.filter(item =>
            (item.codigo || item.apartamento) !== unidade
        );

        if (novo.length === historico.length) return false;

        gravarHistorico(novo);
        registrarEvento(
            "BAIXA",
            "X chegou a 0. Unidade removida automaticamente do histórico.",
            unidade
        );
        return true;
    }

    function limparSessao() {
        unidadeAtual = "";
        quantidadeAtual = 0;
        sessaoAtiva = false;
        filtroValido = false;
        saidaProcessada = false;
        atualizarPainel();
    }

    function sincronizar() {
        if (location.href !== ultimaURL) {
            salvarPendencia("Saiu da página de encomendas");
            ultimaURL = location.href;
            limparSessao();
            return;
        }

        const modo = modoFiltro();

        if (modo === "todos" || modo === "entregues") {
            atualizarPainel();
            return;
        }

        if (modo === "outro") {
            atualizarPainel();
            return;
        }

        filtroValido = true;

        const unidade = lerUnidadeSelecionada();

        if (!unidade) {
            if (sessaoAtiva && quantidadeAtual > 0) {
                salvarPendencia("Campo apagado antes de concluir a baixa");
            }
            limparSessao();
            return;
        }

        const x = lerX();
        if (x === null) return;

        if (
            sessaoAtiva &&
            unidadeAtual &&
            unidade !== unidadeAtual &&
            quantidadeAtual > 0
        ) {
            salvarPendencia(
                "Outra residência foi pesquisada antes de concluir a baixa"
            );
            limparSessao();
            filtroValido = true;
        }

        unidadeAtual = unidade;
        quantidadeAtual = x;
        saidaProcessada = false;

        if (x === 0) {
            sessaoAtiva = false;
            removerDoHistorico(unidade);
            atualizarPainel();
            return;
        }

        sessaoAtiva = true;
        atualizarPainel();
    }

    document.addEventListener("click", evento => {
        const botao = evento.target.closest("button");
        if (!botao) return;

        if (
            botao.getAttribute("data-testid") ===
            "residence-autocomplete-clear-input-button"
        ) {
            salvarPendencia("Campo apagado antes de concluir a baixa");
            limparSessao();
        }
    }, true);

    function processarFechamento() {
        salvarPendencia("Página fechada ou atualizada");
    }

    window.addEventListener("beforeunload", processarFechamento, {
        capture: true
    });
    window.addEventListener("pagehide", processarFechamento, {
        capture: true
    });

    function adicionarEstilo() {
        if (document.querySelector("#cssMonitor505")) return;

        const style = document.createElement("style");
        style.id = "cssMonitor505";
        style.textContent = `
            #painelConsultas505,#painelConsultas505 *{box-sizing:border-box;font-family:Arial,Helvetica,sans-serif}
            #painelConsultas505{animation:abrir505 .25s ease-out}
            @keyframes abrir505{from{opacity:0;transform:translateY(-10px) scale(.98)}to{opacity:1;transform:none}}
            @keyframes led505{50%{opacity:.3;transform:scale(.8)}}
            .conteudo505{position:relative;z-index:2;padding:18px;height:100%;display:flex;flex-direction:column}
            .topo505{display:flex;justify-content:space-between;align-items:center}
            .titulo505{font-size:18px;font-weight:bold;color:#22c55e;letter-spacing:1px;text-shadow:0 0 15px rgba(34,197,94,.9)}
            .subtitulo505{font-size:10px;color:#777;margin-top:5px;letter-spacing:2px}
            .led505{width:13px;height:13px;border-radius:50%;background:#22c55e;box-shadow:0 0 18px #22c55e;animation:led505 1.3s infinite}
            .status505{margin-top:15px;padding:12px;border-radius:12px;background:rgba(0,0,0,.72);border:1px solid rgba(34,197,94,.25);font-size:12px;color:#aaa;line-height:20px}
            .linha505{display:flex;justify-content:space-between;gap:10px}.valor505{color:#22c55e;font-weight:bold;text-align:right}
            .acoes505{display:flex;gap:8px;margin-top:10px}.acoes505 button{flex:1;padding:8px;border-radius:9px;font-weight:bold;font-size:10px;cursor:pointer}
            #limpar505{background:rgba(239,68,68,.12);border:1px solid #ef4444;color:#ef4444}
            #fechar505{background:rgba(34,197,94,.12);border:1px solid #22c55e;color:#22c55e}
            .lista505{flex:1;min-height:0;overflow-y:auto;margin-top:12px;padding-right:5px}
            .registro505{background:rgba(17,24,39,.88);border:1px solid rgba(34,197,94,.2);padding:12px;border-radius:12px;margin-bottom:9px}
            .codigo505{font-size:18px;font-weight:bold;color:#22c55e}.data505{margin-top:5px;font-size:11px;color:#888}.motivo505{margin-top:5px;font-size:10px;color:#fbbf24;line-height:15px}
            .vazio505{text-align:center;margin-top:80px;color:#666;font-size:12px;letter-spacing:2px}
            .rodape505{text-align:center;font-size:9px;color:#555;padding-top:8px}
            #botaoMonitor505{display:inline-flex!important;align-items:center;justify-content:center;gap:8px;white-space:nowrap;margin-left:10px!important}
            #tooltipMonitor505{display:none;position:fixed;width:320px;padding:15px;background:#050505;border:1px solid #22c55e;border-radius:12px;z-index:2147483647;box-shadow:0 0 30px rgba(34,197,94,.35);color:#fff}
            @media(max-width:600px){#painelConsultas505{top:10px!important;left:10px!important;right:10px!important;width:auto!important;height:calc(100vh - 20px)!important}}
        `;
        (document.head || document.documentElement).appendChild(style);
    }

    function criarBotao() {
        adicionarEstilo();
        if (document.querySelector("#botaoMonitor505")) return;

        const referencia = document.querySelector(
            'button[data-testid="delivery-select-multiple-deliveries-button"]'
        );
        if (!referencia) return;

        const botao = document.createElement("button");
        botao.id = "botaoMonitor505";
        botao.type = "button";
        botao.className = referencia.className;
        botao.innerHTML = 'MONITOR DE ENCOMENDAS <span class="led505" style="width:8px;height:8px"></span>';
        botao.onclick = abrirPainel;
        referencia.insertAdjacentElement("afterend", botao);

        const tooltip = document.createElement("div");
        tooltip.id = "tooltipMonitor505";
        tooltip.innerHTML = `
            <div style="color:#22c55e;font-weight:bold;margin-bottom:10px">MONITOR DE ENCOMENDAS</div>
            <div style="font-size:12px;color:#aaa;line-height:18px">
                Monitora somente <b style="color:#22c55e">Aguardando entrega</b>.<br><br>
                Todos e Entregues apenas pausam a leitura, sem apagar a sessão válida.<br><br>
                X = 0 remove a unidade do histórico. X maior que 0 só é registrado ao sair, apagar o campo, atualizar ou fechar.
            </div>
            <div style="margin-top:10px;text-align:right;color:#22c55e;font-size:11px">Criado por Daniel Alexandre</div>`;
        document.body.appendChild(tooltip);

        botao.onmouseenter = () => {
            const p = botao.getBoundingClientRect();
            tooltip.style.display = "block";
            tooltip.style.top = `${p.bottom + 8}px`;
            tooltip.style.left = `${Math.max(10, Math.min(p.left, innerWidth - 330))}px`;
        };
        botao.onmouseleave = () => tooltip.style.display = "none";
    }

    function abrirPainel() {
        const antigo = document.querySelector("#painelConsultas505");
        if (antigo) {
            antigo.remove();
            if (intervaloMatrix) clearInterval(intervaloMatrix);
            intervaloMatrix = null;
            return;
        }

        adicionarEstilo();
        const painel = document.createElement("div");
        painel.id = "painelConsultas505";
        Object.assign(painel.style, {
            position: "fixed",
            top: "60px",
            right: "25px",
            width: "410px",
            height: "610px",
            background: "#050505",
            color: "#fff",
            zIndex: "2147483646",
            borderRadius: "18px",
            overflow: "hidden",
            border: "1px solid #22c55e",
            boxShadow: "0 0 35px rgba(34,197,94,.35)"
        });

        painel.innerHTML = `
            <canvas id="matrix505" style="position:absolute;inset:0;width:100%;height:100%;opacity:.12;pointer-events:none"></canvas>
            <div class="conteudo505">
                <div class="topo505"><div><div class="titulo505">MONITOR DE ENCOMENDAS</div><div class="subtitulo505">SYSTEM ONLINE · V505.2</div></div><div class="led505"></div></div>
                <div class="status505">
                    <div class="linha505"><span>FILTRO</span><span id="filtro505" class="valor505">AGUARDANDO</span></div>
                    <div class="linha505"><span>UNIDADE</span><span id="unidade505" class="valor505">NENHUMA</span></div>
                    <div class="linha505"><span>ETAPA</span><span id="etapa505" class="valor505">AGUARDANDO</span></div>
                    <div class="linha505"><span>REGISTROS</span><span id="registros505" class="valor505">0</span></div>
                    <div class="linha505"><span>ÚLTIMA AÇÃO</span><span id="acao505" class="valor505">SISTEMA INICIADO</span></div>
                </div>
                <div class="acoes505"><button id="limpar505">LIMPAR HISTÓRICO</button><button id="fechar505">FECHAR</button></div>
                <div id="lista505" class="lista505"></div>
                <div class="rodape505">Criado por Daniel Alexandre</div>
            </div>`;

        document.body.appendChild(painel);
        painel.querySelector("#fechar505").onclick = () => abrirPainel();
        painel.querySelector("#limpar505").onclick = () => {
            if (!confirm("Deseja realmente limpar todo o histórico?")) return;
            localStorage.removeItem(STORAGE_HISTORICO);
            atualizarPainel();
        };
        atualizarPainel();
        iniciarMatrix();
    }

    function atualizarPainel() {
        const painel = document.querySelector("#painelConsultas505");
        if (!painel) return;

        const modo = modoFiltro();
        const historico = obterHistorico();
        const pausado = modo === "todos" || modo === "entregues";

        painel.querySelector("#filtro505").textContent =
            modo === "aguardando" ? "AGUARDANDO ENTREGA" : pausado ? "PAUSADO" : "FORA DA TELA";
        painel.querySelector("#unidade505").textContent =
            modo === "aguardando" && unidadeAtual ? unidadeAtual : "NENHUMA";
        painel.querySelector("#etapa505").textContent = pausado
            ? "PAUSADO"
            : modo === "aguardando" && unidadeAtual
                ? quantidadeAtual > 0 ? `${quantidadeAtual} RESTANTE(S)` : "SEM PENDÊNCIAS"
                : "AGUARDANDO";
        painel.querySelector("#registros505").textContent = String(historico.length);
        painel.querySelector("#acao505").textContent = logsSistema[0]
            ? `${logsSistema[0].tipo}: ${logsSistema[0].mensagem}`
            : "SISTEMA INICIADO";

        painel.querySelector("#lista505").innerHTML = historico.length
            ? historico.map(item => `
                <div class="registro505">
                    <div class="codigo505">${escaparHTML(item.codigo || item.apartamento)}</div>
                    <div class="data505">${escaparHTML(item.dataCompleta || "")}</div>
                    <div class="motivo505">${escaparHTML(item.motivo || "Possível baixa não confirmada")}</div>
                </div>`).join("")
            : '<div class="vazio505">SEM REGISTROS</div>';
    }

    function iniciarMatrix() {
        const canvas = document.querySelector("#matrix505");
        const painel = document.querySelector("#painelConsultas505");
        if (!canvas || !painel) return;

        canvas.width = painel.offsetWidth;
        canvas.height = painel.offsetHeight;
        const ctx = canvas.getContext("2d");
        if (!ctx) return;

        const chars = "01ABCDEFGHIJKLMNOPQRSTUVWXYZ#$%&@";
        const tam = 14;
        const gotas = Array(Math.floor(canvas.width / tam)).fill(0).map(() => Math.random() * 40);

        if (intervaloMatrix) clearInterval(intervaloMatrix);
        intervaloMatrix = setInterval(() => {
            if (!document.body.contains(canvas)) {
                clearInterval(intervaloMatrix);
                intervaloMatrix = null;
                return;
            }
            ctx.fillStyle = "rgba(0,0,0,.12)";
            ctx.fillRect(0, 0, canvas.width, canvas.height);
            ctx.fillStyle = "#22c55e";
            ctx.font = `${tam}px monospace`;
            gotas.forEach((y, i) => {
                ctx.fillText(chars[Math.floor(Math.random() * chars.length)], i * tam, y * tam);
                if (y * tam > canvas.height && Math.random() > .975) gotas[i] = 0;
                gotas[i]++;
            });
        }, 90);
    }

    new MutationObserver(criarBotao).observe(document.documentElement, {
        childList: true,
        subtree: true
    });

    setInterval(criarBotao, 1500);
    setInterval(sincronizar, INTERVALO);
    setInterval(atualizarPainel, 500);

    function iniciar() {
        adicionarEstilo();
        criarBotao();
        sincronizar();
        registrarEvento("SISTEMA", "Monitor v505.2 iniciado", "MONITORAMENTO ATIVO");
    }

    if (document.readyState === "loading") {
        document.addEventListener("DOMContentLoaded", iniciar, { once: true });
    } else {
        iniciar();
    }
})();
