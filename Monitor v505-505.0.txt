(function () {

    "use strict";


    // ==========================================================
    // CONFIGURAÇÕES
    // ==========================================================

    const STORAGE_ATUAL =
        "monitor503_atual";

    const STORAGE_HISTORICO =
        "monitor503_historico";

    const STORAGE_FLUXO =
        "monitor505_fluxo";


    // ==========================================================
    // VARIÁVEIS
    // ==========================================================

    let encomendaAtual =
        localStorage.getItem(STORAGE_ATUAL) || "";

    let ultimoValor =
        encomendaAtual || "";

    let ultimaURL =
        location.href;

    let logsSistema =
        [];

    let intervaloMatrix =
        null;

    let baixaIniciada =
        sessionStorage.getItem(STORAGE_FLUXO) === "1";

    let baixaConfirmada =
        false;

    let processandoSaida =
        false;


    // ==========================================================
    // UTILITÁRIOS
    // ==========================================================

    function escaparHTML(valor) {

        return String(valor ?? "")
            .replaceAll("&", "&amp;")
            .replaceAll("<", "&lt;")
            .replaceAll(">", "&gt;")
            .replaceAll('"', "&quot;")
            .replaceAll("'", "&#039;");

    }


    function normalizarTexto(valor) {

        return String(valor || "")
            .trim()
            .toLowerCase()
            .normalize("NFD")
            .replace(/[\u0300-\u036f]/g, "");

    }


    function registrarEvento(
        tipo,
        mensagem,
        destaque = ""
    ) {

        const agora =
            new Date();

        logsSistema.unshift({

            tipo,
            mensagem,
            destaque,

            hora:
                agora.toLocaleTimeString("pt-BR")

        });

        logsSistema =
            logsSistema.slice(0, 100);

        console.log(
            `[MONITOR v505] ${tipo}:`,
            mensagem,
            destaque
        );

        atualizarPainel505();

    }


    // ==========================================================
    // HISTÓRICO
    // ==========================================================

    function obterHistorico() {

        try {

            const lista =
                JSON.parse(
                    localStorage.getItem(
                        STORAGE_HISTORICO
                    ) || "[]"
                );

            return Array.isArray(lista)
                ? lista
                : [];

        } catch (erro) {

            console.error(
                "Erro ao ler histórico:",
                erro
            );

            return [];

        }

    }


    function salvarHistorico(
        codigo,
        motivo = "Possível abandono"
    ) {

        codigo =
            String(codigo || "").trim();

        if (!codigo)
            return;

        let lista =
            obterHistorico();

        const agora =
            new Date();

        const ultimo =
            lista[0];

        /*
         * Evita salvar o mesmo código duas vezes
         * em poucos segundos.
         */

        if (
            ultimo &&
            ultimo.codigo === codigo &&
            ultimo.timestamp &&
            Date.now() -
            Number(ultimo.timestamp) < 3000
        ) {

            return;

        }

        lista.unshift({

            codigo,

            data:
                agora.toLocaleDateString("pt-BR"),

            hora:
                agora.toLocaleTimeString("pt-BR"),

            dataCompleta:
                agora.toLocaleString("pt-BR"),

            timestamp:
                Date.now(),

            motivo

        });

        lista =
            lista.slice(0, 1000);

        localStorage.setItem(
            STORAGE_HISTORICO,
            JSON.stringify(lista)
        );

        registrarEvento(
            "HISTÓRICO",
            motivo,
            codigo
        );

    }


    // ==========================================================
    // ENCOMENDA ATUAL
    // ==========================================================

    function salvarAtual(valor) {

        valor =
            String(valor || "").trim();

        if (!valor)
            return;

        if (!/^\d+\/\d+$/.test(valor))
            return;

        if (valor === ultimoValor)
            return;

        ultimoValor =
            valor;

        encomendaAtual =
            valor;

        baixaIniciada =
            false;

        baixaConfirmada =
            false;

        processandoSaida =
            false;

        sessionStorage.removeItem(
            STORAGE_FLUXO
        );

        localStorage.setItem(
            STORAGE_ATUAL,
            valor
        );

        registrarEvento(
            "CAPTURA",
            "Encomenda identificada",
            valor
        );

    }


    function limparAtual(
        motivo = "",
        registrar = true
    ) {

        const codigoAnterior =
            encomendaAtual;

        encomendaAtual =
            "";

        ultimoValor =
            "";

        localStorage.removeItem(
            STORAGE_ATUAL
        );

        if (
            registrar &&
            codigoAnterior
        ) {

            registrarEvento(
                "LIMPEZA",
                motivo ||
                "Encomenda atual removida",
                codigoAnterior
            );

        }

        atualizarPainel505();

    }


    // ==========================================================
    // FLUXO DA BAIXA
    // ==========================================================

    function iniciarBaixa() {

        baixaIniciada =
            true;

        baixaConfirmada =
            false;

        processandoSaida =
            false;

        sessionStorage.setItem(
            STORAGE_FLUXO,
            "1"
        );

        registrarEvento(
            "BAIXA",
            "Encerramento iniciado",
            encomendaAtual ||
            "SEM CÓDIGO"
        );

    }


    function concluirBaixa() {

        const codigo =
            encomendaAtual;

        /*
         * O clique no botão SIM é a única
         * confirmação final válida.
         */

        baixaConfirmada =
            true;

        baixaIniciada =
            false;

        processandoSaida =
            true;

        sessionStorage.removeItem(
            STORAGE_FLUXO
        );

        /*
         * Limpa antes que o e-Condos mude
         * de rota ou atualize a tela.
         */

        limparAtual(
            "",
            false
        );

        registrarEvento(
            "BAIXA CONCLUÍDA",
            "Clique final em Sim confirmado. Não enviado ao histórico.",
            codigo
        );

    }


    function registrarAbandono(motivo) {

        if (!encomendaAtual)
            return;

        if (baixaConfirmada)
            return;

        if (processandoSaida)
            return;

        processandoSaida =
            true;

        const codigo =
            encomendaAtual;

        salvarHistorico(
            codigo,
            motivo
        );

        baixaIniciada =
            false;

        sessionStorage.removeItem(
            STORAGE_FLUXO
        );

        limparAtual(
            "",
            false
        );

        registrarEvento(
            "ABANDONO",
            motivo,
            codigo
        );

    }// ==========================================================
    // CAPTURA DO CAMPO
    // ==========================================================

    function procurarEncomenda() {

        const campos =
            document.querySelectorAll("input");

        campos.forEach(campo => {

            if (campo.offsetParent === null)
                return;

            const valor =
                String(
                    campo.value || ""
                ).trim();

            if (/^\d+\/\d+$/.test(valor)) {

                salvarAtual(valor);

            }

        });

    }


    document.addEventListener(
        "input",
        evento => {

            const campo =
                evento.target;

            if (
                !(campo instanceof HTMLInputElement)
            ) {

                return;

            }

            const valor =
                String(
                    campo.value || ""
                ).trim();

            if (/^\d+\/\d+$/.test(valor)) {

                salvarAtual(valor);

            }

        },
        true
    );


    document.addEventListener(
        "change",
        evento => {

            const campo =
                evento.target;

            if (
                !(campo instanceof HTMLInputElement)
            ) {

                return;

            }

            const valor =
                String(
                    campo.value || ""
                ).trim();

            if (/^\d+\/\d+$/.test(valor)) {

                salvarAtual(valor);

            }

        },
        true
    );


    const observadorCampo =
        new MutationObserver(() => {

            procurarEncomenda();
            criarBotaoMonitor();

        });


    observadorCampo.observe(
        document.documentElement,
        {
            childList: true,
            subtree: true
        }
    );


    setInterval(() => {

        procurarEncomenda();

    }, 500);


    // ==========================================================
    // IDENTIFICAÇÃO DOS BOTÕES
    // ==========================================================

    function eBotaoEncerrar(botao) {

        return (
            botao.getAttribute(
                "data-testid"
            ) ===
            "delivery-close-button"
        );

    }


    function eBotaoSimFinal(botao) {

        if (
            !botao.classList.contains(
                "swal2-confirm"
            )
        ) {

            return false;

        }

        if (
            !botao.classList.contains(
                "swal2-styled"
            )
        ) {

            return false;

        }

        return normalizarTexto(
            botao.textContent
        ) === "sim";

    }


    function eBotaoCancelar(botao) {

        return (
            botao.classList.contains(
                "swal2-cancel"
            ) ||
            botao.classList.contains(
                "swal2-close"
            )
        );

    }


    function eBotaoDoMonitor(botao) {

        return Boolean(
            botao.id ===
                "botaoMonitor505"
            ||
            botao.closest(
                "#painelConsultas505"
            )
        );

    }


    function pareceBotaoLixeira(botao) {

        if (!botao)
            return false;

        if (eBotaoDoMonitor(botao))
            return false;

        if (eBotaoEncerrar(botao))
            return false;

        if (eBotaoSimFinal(botao))
            return false;

        if (eBotaoCancelar(botao))
            return false;


        /*
         * Durante o fluxo da baixa,
         * nenhum botão genérico com SVG
         * será tratado como lixeira.
         */

        if (baixaIniciada)
            return false;


        const testId =
            normalizarTexto(
                botao.getAttribute(
                    "data-testid"
                )
            );

        const ariaLabel =
            normalizarTexto(
                botao.getAttribute(
                    "aria-label"
                )
            );

        const titulo =
            normalizarTexto(
                botao.getAttribute(
                    "title"
                )
            );

        const texto =
            normalizarTexto(
                botao.textContent
            );


        const termos = [

            "delete",
            "remove",
            "clear",
            "excluir",
            "remover",
            "limpar",
            "lixeira"

        ];


        const identificadoPorNome =
            termos.some(termo => {

                return (
                    testId.includes(termo)
                    ||
                    ariaLabel.includes(termo)
                    ||
                    titulo.includes(termo)
                    ||
                    texto === termo
                );

            });


        if (identificadoPorNome)
            return true;


        const svg =
            botao.querySelector("svg");

        if (!svg)
            return false;


        const html =
            svg.outerHTML.toLowerCase();


        return (
            html.includes("currentcolor")
            &&
            html.includes("<path")
        );

    }


    // ==========================================================
    // CONTROLE DOS CLIQUES
    // ==========================================================

    document.addEventListener(
        "click",
        evento => {

            const botao =
                evento.target.closest("button");

            if (!botao)
                return;

            if (eBotaoDoMonitor(botao))
                return;


            // --------------------------------------------------
            // PASSO 3 — CLIQUE FINAL EM SIM
            // --------------------------------------------------

            if (eBotaoSimFinal(botao)) {

                if (
                    encomendaAtual &&
                    baixaIniciada
                ) {

                    concluirBaixa();

                } else {

                    registrarEvento(
                        "AVISO",
                        "Botão Sim detectado fora do fluxo monitorado"
                    );

                }

                return;

            }


            // --------------------------------------------------
            // PASSO 1 — BOTÃO ENCERRAR
            // --------------------------------------------------

            if (eBotaoEncerrar(botao)) {

                iniciarBaixa();

                return;

            }


            // --------------------------------------------------
            // CANCELAMENTO DA CONFIRMAÇÃO
            // --------------------------------------------------

            if (eBotaoCancelar(botao)) {

                if (baixaIniciada) {

                    registrarEvento(
                        "BAIXA",
                        "Confirmação final cancelada",
                        encomendaAtual
                    );

                    baixaIniciada =
                        false;

                    sessionStorage.removeItem(
                        STORAGE_FLUXO
                    );

                }

                return;

            }


            // --------------------------------------------------
            // LIXEIRA / LIMPEZA DO CAMPO
            // --------------------------------------------------

            if (pareceBotaoLixeira(botao)) {

                if (encomendaAtual) {

                    registrarAbandono(
                        "Campo limpo sem confirmação final"
                    );

                } else {

                    registrarEvento(
                        "AÇÃO",
                        "Botão de limpeza detectado sem encomenda ativa"
                    );

                }

            }

        },
        true
    );


    // ==========================================================
    // TROCA DE PÁGINA
    // ==========================================================

    setInterval(() => {

        if (location.href === ultimaURL)
            return;


        const urlAnterior =
            ultimaURL;

        ultimaURL =
            location.href;


        registrarEvento(
            "NAVEGAÇÃO",
            "Mudança de página detectada",
            `${urlAnterior} → ${ultimaURL}`
        );


        /*
         * Se já confirmou no botão Sim,
         * a mudança de página é normal.
         */

        if (baixaConfirmada)
            return;


        /*
         * Se ainda existe encomenda ativa,
         * qualquer saída sem confirmação final
         * será registrada.
         */

        if (encomendaAtual) {

            registrarAbandono(

                baixaIniciada

                    ? "Saiu da página antes de confirmar o passo final"

                    : "Página abandonada sem concluir a baixa"

            );

        }

    }, 500);


    // ==========================================================
    // FECHAR, RECARREGAR OU SAIR DO SISTEMA
    // ==========================================================

    function processarFechamento() {

        if (!encomendaAtual)
            return;

        if (baixaConfirmada)
            return;

        if (processandoSaida)
            return;


        const codigo =
            encomendaAtual;

        const motivo =
            baixaIniciada

                ? "Sistema fechado ou atualizado antes do clique final em Sim"

                : "Sistema fechado ou atualizado sem concluir a baixa";


        let lista =
            obterHistorico();

        const agora =
            new Date();


        lista.unshift({

            codigo,

            data:
                agora.toLocaleDateString("pt-BR"),

            hora:
                agora.toLocaleTimeString("pt-BR"),

            dataCompleta:
                agora.toLocaleString("pt-BR"),

            timestamp:
                Date.now(),

            motivo

        });


        lista =
            lista.slice(0, 1000);


        localStorage.setItem(
            STORAGE_HISTORICO,
            JSON.stringify(lista)
        );


        localStorage.removeItem(
            STORAGE_ATUAL
        );


        sessionStorage.removeItem(
            STORAGE_FLUXO
        );


        processandoSaida =
            true;

    }


    window.addEventListener(
        "beforeunload",
        processarFechamento,
        {
            capture: true
        }
    );


    window.addEventListener(
        "pagehide",
        processarFechamento,
        {
            capture: true
        }

        );

    function configurarEventosPainel505(
        painel
    ) {

        const abaHistorico =
            painel.querySelector(
                "#abaHistorico505"
            );


        const abaTempoReal =
            painel.querySelector(
                "#abaTempoReal505"
            );


        const listaHistorico =
            painel.querySelector(
                "#listaHistorico505"
            );


        const consoleTempoReal =
            painel.querySelector(
                "#consoleTempoReal505"
            );


        abaHistorico.onclick = () => {

            painel.dataset.aba =
                "historico";


            abaHistorico.classList.add(
                "ativa505"
            );


            abaTempoReal.classList.remove(
                "ativa505"
            );


            listaHistorico.style.display =
                "block";


            consoleTempoReal.style.display =
                "none";

        };


        abaTempoReal.onclick = () => {

            painel.dataset.aba =
                "tempo-real";


            abaTempoReal.classList.add(
                "ativa505"
            );


            abaHistorico.classList.remove(
                "ativa505"
            );


            listaHistorico.style.display =
                "none";


            consoleTempoReal.style.display =
                "block";


            atualizarPainel505();

        };


        painel.querySelector(
            "#fecharPainel505"
        ).onclick = () => {

            painel.remove();


            if (intervaloMatrix) {

                clearInterval(
                    intervaloMatrix
                );


                intervaloMatrix =
                    null;

            }

        };


        painel.querySelector(
            "#limparHistorico505"
        ).onclick = () => {

            if (

                confirm(
                    "Deseja realmente limpar todo o histórico de encomendas?"
                )

            ) {

                localStorage.removeItem(
                    STORAGE_HISTORICO
                );


                registrarEvento(
                    "HISTÓRICO",
                    "Histórico apagado manualmente"
                );


                atualizarPainel505();

            }

        };


        painel.querySelector(
            "#limparConsole505"
        ).onclick = () => {

            logsSistema =
                [];


            registrarEvento(
                "SISTEMA",
                "Console limpo"
            );


            atualizarPainel505();

        };

    }

    // ATUALIZAR PAINEL
    // ==========================================================

    function atualizarPainel505() {

        const painel =
            document.querySelector(
                "#painelConsultas505"
            );


        if (!painel)
            return;


        const historico =
            obterHistorico();


        const codigoAtual =
            painel.querySelector(
                "#codigoAtual505"
            );


        const etapaBaixa =
            painel.querySelector(
                "#etapaBaixa505"
            );


        const quantidade =
            painel.querySelector(
                "#quantidadeRegistros505"
            );


        const ultimaAcao =
            painel.querySelector(
                "#ultimaAcao505"
            );


        if (codigoAtual) {

            codigoAtual.textContent =
                encomendaAtual ||
                "NENHUMA";

        }


        if (etapaBaixa) {

            if (baixaConfirmada) {

                etapaBaixa.textContent =
                    "FINALIZADA";

            } else if (baixaIniciada) {

                etapaBaixa.textContent =
                    "AGUARDANDO SIM";

            } else if (encomendaAtual) {

                etapaBaixa.textContent =
                    "CÓDIGO CAPTURADO";

            } else {

                etapaBaixa.textContent =
                    "AGUARDANDO";

            }

        }


        if (quantidade) {

            quantidade.textContent =
                String(
                    historico.length
                );

        }


        if (ultimaAcao) {

            const ultimoLog =
                logsSistema[0];


            ultimaAcao.textContent =

                ultimoLog

                    ? (
                        ultimoLog.destaque
                        ||
                        `${ultimoLog.tipo}: ${ultimoLog.mensagem}`
                    )

                    : "SEM AÇÕES";

        }


        // ======================================================
        // LISTA DO HISTÓRICO
        // ======================================================

        const listaHistorico =
            painel.querySelector(
                "#listaHistorico505"
            );


        if (listaHistorico) {

            if (!historico.length) {

                listaHistorico.innerHTML = `

                    <div class="vazioSistema505">

                        SEM REGISTROS

                    </div>

                `;

            } else {

                listaHistorico.innerHTML =

                    historico.map(item => `

                        <div class="registroSistema505">


                            <div class="codigoSistema505">

                                ${escaparHTML(
                                    item.codigo
                                )}

                            </div>


                            <div class="dataSistema505">

                                ${escaparHTML(

                                    item.dataCompleta

                                    ||

                                    `${item.data || ""} - ${item.hora || ""}`

                                )}

                            </div>


                            <div class="motivoSistema505">

                                ${escaparHTML(

                                    item.motivo

                                    ||

                                    "Possível baixa não confirmada"

                                )}

                            </div>


                        </div>

                    `).join("");

            }

        }


        // ======================================================
        // CONSOLE EM TEMPO REAL
        // ======================================================

        const consoleTempoReal =
            painel.querySelector(
                "#consoleTempoReal505"
            );


        if (consoleTempoReal) {

            if (!logsSistema.length) {

                consoleTempoReal.innerHTML = `

                    <div class="vazioSistema505">

                        SEM ATIVIDADES

                    </div>

                `;

            } else {

                consoleTempoReal.innerHTML =

                    logsSistema.map(log => `

                        <div class="eventoConsole505">


                            <div class="eventoTopo505">


                                <span class="eventoTipo505">

                                    ${escaparHTML(
                                        log.tipo
                                    )}

                                </span>


                                <span class="eventoHora505">

                                    ${escaparHTML(
                                        log.hora
                                    )}

                                </span>


                            </div>


                            <div class="eventoMensagem505">


                                ${escaparHTML(
                                    log.mensagem
                                )}


                                ${
                                    log.destaque

                                        ? `

                                            <span class="eventoDestaque505">

                                                · ${escaparHTML(
                                                    log.destaque
                                                )}

                                            </span>

                                        `

                                        : ""
                                }


                            </div>


                        </div>

                    `).join("");

            }

        }

    }

    function matrixPainel505() {

        const canvas =
            document.querySelector(
                "#matrixPainel505"
            );


        const painel =
            document.querySelector(
                "#painelConsultas505"
            );


        if (
            !canvas ||
            !painel
        ) {

            return;

        }


        canvas.width =
            painel.offsetWidth;


        canvas.height =
            painel.offsetHeight;


        const contexto =
            canvas.getContext("2d");


        if (!contexto)
            return;


        const caracteres =
            "01ABCDEFGHIJKLMNOPQRSTUVWXYZ#$%&@";


        const tamanho =
            14;


        const colunas =
            Math.floor(
                canvas.width /
                tamanho
            );


        const gotas =
            [];


        for (

            let indice = 0;

            indice < colunas;

            indice++

        ) {

            gotas[indice] =
                Math.random() * 40;

        }


        if (intervaloMatrix) {

            clearInterval(
                intervaloMatrix
            );

        }


        intervaloMatrix =
            setInterval(() => {

                if (

                    !document.body.contains(
                        canvas
                    )

                ) {

                    clearInterval(
                        intervaloMatrix
                    );


                    intervaloMatrix =
                        null;


                    return;

                }


                contexto.fillStyle =
                    "rgba(0,0,0,.12)";


                contexto.fillRect(

                    0,

                    0,

                    canvas.width,

                    canvas.height

                );


                contexto.fillStyle =
                    "#22c55e";


                contexto.font =
                    `${tamanho}px monospace`;


                for (

                    let indice = 0;

                    indice < gotas.length;

                    indice++

                ) {

                    const caractere =

                        caracteres[

                            Math.floor(

                                Math.random() *

                                caracteres.length

                            )

                        ];


                    contexto.fillText(

                        caractere,

                        indice * tamanho,

                        gotas[indice] *
                        tamanho

                    );


                    if (

                        gotas[indice] *
                        tamanho >

                        canvas.height

                        &&

                        Math.random() >
                        .975

                    ) {

                        gotas[indice] =
                            0;

                    }


                    gotas[indice]++;

                }

            }, 90);

    }
